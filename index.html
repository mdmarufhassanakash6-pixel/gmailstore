import os import telebot import google.generativeai as genai # রেন্ডার বা রেলওয়ের এনভায়রনমেন্ট ভেরিয়েবল থেকে টোকেন ও এআই কি রিড করা TELEGRAM_TOKEN = os.getenv('TELEGRAM_TOKEN') ADMIN_ID = 8357226129 GEMINI_API_KEY = os.getenv('GEMINI_API_KEY') bot = telebot.TeleBot(TELEGRAM_TOKEN) genai.configure(api_key=GEMINI_API_KEY) model = genai.GenerativeModel("gemini-1.5-flash") @bot.message_handler(commands=['start']) def send_welcome(message): bot.reply_to(message, "আসসালামু আলাইকুম! জিমেইল বাই-সেল শপে আপনাকে স্বাগতম।") # কাস্টমারের মেসেজ হ্যান্ডেল করা এবং অ্যাডমিনকে নোটিফিকেশন পাঠানো @bot.message_handler(func=lambda message: message.from_user.id != ADMIN_ID) def handle_customer(message): user_id = message.chat.id username = message.from_user.first_name or "User" text = message.text # অ্যাডমিন ইনবক্সে কাস্টমারের মেসেজ পাঠানোর কোড try: notification = ( f"📥 **নতুন কাস্টমার মেসেজ!**\n\n" f"👤 নাম: {username}\n" f"🆔 ইউজার আইডি: `{user_id}`\n" f"💬 মেসেজ: {text}" ) bot.send_message(ADMIN_ID, notification, parse_mode="Markdown") except Exception as e: print(f"Notification Error: {e}") # জেমিনি এআই প্রম্পট ও অটো-রিপ্লাই জেনারেট করা prompt = f""" You are a professional and smart AI customer support assistant for a Gmail buying/selling shop. You must reply in the exact language the user writes in (Bangla or English). Be polite, concise, and helpful. Answer the customer's query based on general Gmail shop services. Customer Message: {text} """ try: response = model.generate_content(prompt) reply_text = response.text bot.reply_to(message, reply_text) except Exception as e: bot.reply_to(message, "দুঃখিত, এই মুহূর্তে উত্তর দিতে সমস্যা হচ্ছে। একটু পরে আবার চেষ্টা করুন।") print(f"Gemini Error: {e}") # অ্যাডমিনের রিপ্লাই হ্যান্ডেল করার কোড @bot.message_handler(func=lambda message: message.from_user.id == ADMIN_ID and message.reply_to_message is not None) def handle_admin_reply(message): try: replied_msg = message.reply_to_message.text # নোটিফিকেশন থেকে ইউজারের আইডি বের করা if "🆔 ইউজার আইডি:" in replied_msg: lines = replied_msg.split("\n") for line in lines: if "🆔 ইউজার আইডি:" in line: target_user_id = int(line.split("`")[1]) bot.send_message(target_user_id, f"অ্যাডমিনের উত্তর:\n{message.text}") bot.reply_to(message, "✅ কাস্টমারের কাছে মেসেজ সফলভাবে পাঠানো হয়েছে!") return except Exception as e: bot.reply_to(message, f"❌ রিপ্লাই পাঠাতে সমস্যা হয়েছে: {e}") # বট রান করার জন্য ইনফিনিট পোলিং bot.infinity_polling()
